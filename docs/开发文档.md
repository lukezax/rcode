### 3.2 数据表详细设计

> **实现说明**：实际项目使用 SQLite（`sql.js` 驱动）+ TypeORM，开发环境通过 `synchronize: true` 由实体自动建表，无需手动执行 DDL。下文 SQL 为逻辑表结构说明（已按 SQLite 语法书写）。

#### 3.2.1 users 表

```sql
CREATE TABLE users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email VARCHAR(255) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    nickname VARCHAR(100),
    avatar VARCHAR(500),
    role VARCHAR(20) DEFAULT 'user' CHECK (role IN ('user', 'admin')),
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'banned')),
    created_at DATETIME DEFAULT CURRENT_DATETIME,
    updated_at DATETIME DEFAULT CURRENT_DATETIME
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_status ON users(status);
```

**字段说明**：

| 字段 | 类型 | 必填 | 默认值 | 说明 |
|------|------|------|--------|------|
| id | INTEGER | 是 | 自增 | 主键 |
| email | VARCHAR(255) | 是 | - | 邮箱，唯一 |
| password | VARCHAR(255) | 是 | - | 密码（bcrypt 加密） |
| nickname | VARCHAR(100) | 否 | - | 昵称 |
| avatar | VARCHAR(500) | 否 | - | 头像 URL |
| role | VARCHAR(20) | 是 | 'user' | 角色：user/admin |
| status | VARCHAR(20) | 是 | 'active' | 状态：active/banned |
| created_at | DATETIME | 是 | CURRENT_DATETIME | 创建时间 |
| updated_at | DATETIME | 是 | CURRENT_DATETIME | 更新时间 |

#### 3.2.2 targets 表

```sql
CREATE TABLE targets (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(100) NOT NULL,
    slug VARCHAR(100) UNIQUE NOT NULL,
    category VARCHAR(50) NOT NULL,
    logo_url VARCHAR(500),
    website VARCHAR(500),
    description TEXT,
    is_preset BOOLEAN DEFAULT FALSE,
    status VARCHAR(20) DEFAULT 'pending' CHECK (status IN ('pending', 'active', 'rejected', 'disabled')),
    created_by INTEGER REFERENCES users(id),
    approved_by INTEGER REFERENCES users(id),
    approved_at DATETIME,
    reject_reason TEXT,
    created_at DATETIME DEFAULT CURRENT_DATETIME,
    updated_at DATETIME DEFAULT CURRENT_DATETIME
);

CREATE INDEX idx_targets_slug ON targets(slug);
CREATE INDEX idx_targets_category ON targets(category);
CREATE INDEX idx_targets_status ON targets(status);
CREATE INDEX idx_targets_created_by ON targets(created_by);
```

**字段说明**：

| 字段 | 类型 | 必填 | 默认值 | 说明 |
|------|------|------|--------|------|
| id | INTEGER | 是 | 自增 | 主键 |
| name | VARCHAR(100) | 是 | - | 工具名称 |
| slug | VARCHAR(100) | 是 | - | 唯一标识（URL 友好） |
| category | VARCHAR(50) | 是 | - | 分类 |
| logo_url | VARCHAR(500) | 否 | - | Logo URL |
| website | VARCHAR(500) | 否 | - | 官网 URL |
| description | TEXT | 否 | - | 简介 |
| is_preset | BOOLEAN | 是 | FALSE | 是否预置 |
| status | VARCHAR(20) | 是 | 'pending' | 状态 |
| created_by | INTEGER | 否 | - | 申请用户 ID |
| approved_by | INTEGER | 否 | - | 审批管理员 ID |
| approved_at | DATETIME | 否 | - | 审批时间 |

#### 3.2.3 referral_codes 表

```sql
CREATE TABLE referral_codes (

    valid_count INT DEFAULT 0,
    invalid_count INT DEFAULT 0,

    updated_at DATETIME DEFAULT CURRENT_DATETIME

CREATE INDEX idx_referral_codes_target ON referral_codes(target_id);
CREATE INDEX idx_referral_codes_user ON referral_codes(user_id);
CREATE INDEX idx_referral_codes_code ON referral_codes(code);
CREATE INDEX idx_referral_codes_status ON referral_codes(status);
CREATE INDEX idx_referral_codes_created_at ON referral_codes(created_at DESC);
```

| 字段 | 类型 | 必填 | 默认值 | 说明 |

### 3.3 数据库初始化

SQLite 无需创建数据库或安装扩展，数据库文件首次访问时自动生成。项目采用两种初始化方式：

**方式一：启动时自动初始化（推荐）**

后端启动时由 `DataInitService`（`src/common/services/data-init.service.ts`）自动执行：

- 若不存在管理员账号，则创建默认管理员（`admin@example.com` / `admin123`）；
- 按 `slug` 幂等写入 20 个预置目标（已存在则跳过）。

数据库文件路径由 `.env` 的 `DB_DATABASE` 指定，默认：

```
backend/data/ai_referral_platform.db
```

**方式二：手动执行 SQL（可选）**

如需手动插入数据，可使用 SQLite 客户端执行类似语句：

```sql
-- 插入管理员账号（密码为 bcrypt 加密后的值）
INSERT INTO users (email, password, nickname, role, status)
VALUES (
    'admin@example.com',
    '$2b$10$...',  -- bcrypt 加密的密码
    '管理员',
    'admin',
    'active'
);

-- 插入预置目标
INSERT INTO targets (name, slug, category, logo_url, website, description, is_preset, status)
VALUES
    ('OpenCode', 'opencode', 'AI 编程', NULL, 'https://opencode.ai', 'AI 编程助手', 1, 'active'),
    ('Cursor', 'cursor', 'AI 编程', NULL, 'https://cursor.sh', 'AI 代码编辑器', 1, 'active');
    -- ... 更多预置目标
```

### 4.1 API 设计规范

#### 4.1.1 RESTful 规范

- 使用名词复数表示资源：`/api/targets`, `/api/codes`
- 使用 HTTP 方法表示操作：

  - POST：创建
  - PUT：完整更新

  - 400：请求参数错误

```typescript
// src/modules/codes/codes.service.ts

/**
 * 计算邀请码状态
 */
private calculateStatus(validCount: number, invalidCount: number): string {
  if (validCount > invalidCount) {
    return 'valid';
  } else if (invalidCount > validCount) {
    return 'invalid';
  } else {
    return 'pending';
  }
}

/**
 * 更新邀请码状态
 */
private async updateCodeStatus(codeId: number): Promise<void> {
  const code = await this.codeRepository.findOne({
    where: { id: codeId },
  });

  // 检查是否过期
  if (code.expireAt && new Date() > code.expireAt) {
    code.status = 'expired';
  } else {
    code.status = this.calculateStatus(code.validCount, code.invalidCount);
  }

  await this.codeRepository.save(code);
}
```

### 5.2 反馈逻辑

```typescript
// src/modules/feedbacks/feedbacks.service.ts

async createFeedback(
  userId: number,
  codeId: number,
  vote: 'valid' | 'invalid',
  comment?: string,
): Promise<Feedback> {
  // 1. 检查邀请码是否存在
  const code = await this.codeRepository.findOne({
    where: { id: codeId },
  });
  if (!code) {
    throw new NotFoundException('邀请码不存在');
  }

  // 2. 检查是否是自己的邀请码
  if (code.userId === userId) {
    throw new ForbiddenException('不能给自己的邀请码投票');
  }

  // 3. 检查是否已投票
  let feedback = await this.feedbackRepository.findOne({
    where: { referralCodeId: codeId, userId },
  });

  if (feedback) {
    // 修改投票
    const oldVote = feedback.vote;
    feedback.vote = vote;
    feedback.comment = comment;
    await this.feedbackRepository.save(feedback);

    // 更新计数
    if (oldVote === 'valid' && vote === 'invalid') {
      code.validCount -= 1;
      code.invalidCount += 1;
    } else if (oldVote === 'invalid' && vote === 'valid') {
      code.invalidCount -= 1;
      code.validCount += 1;
    }
  } else {
    // 新投票
    feedback = this.feedbackRepository.create({
      referralCodeId: codeId,
      userId,
      vote,
      comment,
    });
    await this.feedbackRepository.save(feedback);

    // 更新计数
    if (vote === 'valid') {
      code.validCount += 1;
    } else {
      code.invalidCount += 1;
    }
  }

  await this.codeRepository.save(code);

  // 更新状态
  await this.codesService.updateCodeStatus(codeId);

  return feedback;
}
```

### 5.3 上传限制检查

```typescript
// src/modules/codes/codes.service.ts

async createCode(userId: number, createCodeDto: CreateCodeDto): Promise<ReferralCode> {
  // 1. 检查目标是否存在且已启用
  const target = await this.targetRepository.findOne({
    where: { id: createCodeDto.targetId, status: 'active' },
  });
  if (!target) {
    throw new NotFoundException('目标不存在或未启用');
  }

  // 2. 检查邀请码是否已存在
  const existingCode = await this.codeRepository.findOne({
    where: { code: createCodeDto.code },
  });
  if (existingCode) {
    throw new BadRequestException('邀请码已存在');
  }

  // 3. 检查每日上传限制
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const todayCount = await this.codeRepository.count({
    where: {
      userId,
      targetId: createCodeDto.targetId,
      createdAt: MoreThanOrEqual(today),
    },
  });
  if (todayCount >= 3) {
    throw new BadRequestException('同一目标每日最多上传 3 条邀请码');
  }

  // 4. 创建邀请码
  const code = this.codeRepository.create({
    ...createCodeDto,
    userId,
    status: 'pending',
  });
  await this.codeRepository.save(code);

  return code;
}
```

### 5.4 邀请码部分隐藏

```typescript
// src/common/utils/code.util.ts

/**
 * 隐藏邀请码中间部分
 * @param code 完整邀请码
 * @returns 隐藏后的邀请码
 */
export function maskCode(code: string): string {
  if (!code || code.length <= 6) {
    return code;
  }
  const start = code.substring(0, 3);
  const end = code.substring(code.length - 3);
  return `${start}***${end}`;
}
```

### 7.2 JWT 认证

```typescript
// src/common/guards/jwt-auth.guard.ts

import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(private jwtService: JwtService) {}

  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const token = this.extractToken(request);

    if (!token) {
      return false;
    }

    try {
      const payload = this.jwtService.verify(token);
      request.user = payload;
      return true;
    } catch {
      return false;
    }
  }

  private extractToken(request: any): string | null {
    const authHeader = request.headers.authorization;
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return null;
    }
    return authHeader.substring(7);
  }
}
```

### 7.3 角色权限

```typescript
// src/common/guards/roles.guard.ts

import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private readonly roles: string[]) {}

  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    return this.roles.includes(user.role);
  }
}

// 使用示例
```
